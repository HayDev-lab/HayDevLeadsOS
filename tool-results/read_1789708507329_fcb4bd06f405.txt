    97→function OrgTab() {
    98→  const { t } = useLocale();
    99→  const queryClient = useQueryClient();
   100→  const settings = useSettings();
   101→  const org = settings.data?.org;
   102→  const [name, setName] = useState(org?.name ?? "");
   103→  const [locale, setLocale] = useState(org?.locale ?? "hy");
   104→  const [timezone, setTimezone] = useState(org?.timezone ?? "Asia/Yerevan");
   105→  const [currency, setCurrency] = useState(org?.currency ?? "AMD");
   106→  // v0.23: org-configurable forecast commit threshold (Setting row, default 60).
   107→  // Derived-value pattern: `savedThreshold` comes straight from the settings
   108→  // query; `thresholdOverride` only exists while the slider is being dragged
   109→  // (or until save+refetch lands) — no effect/ref syncing needed.
   110→  const savedThreshold = settings.data ? readForecastThreshold(settings.data.settings) : 60;
   111→  const [thresholdOverride, setThresholdOverride] = useState<number | null>(null);
   112→  const commitThreshold = thresholdOverride ?? savedThreshold;
   113→  const save = async () => {
   114→    try {
   115→      const res = await fetch("/api/v1/settings", { method: "PATCH", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ org: { name, locale, timezone, currency } }) });
   116→      if (res.ok) toast.success(t("toast.saved")); else toast.error(t("toast.save_failed"));
   117→      settings.refetch();
   118→    } catch { toast.error(t("toast.save_failed")); }
   119→  };
   120→  const saveThreshold = async (value: number) => {
   121→    try {
   122→      const res = await fetch("/api/v1/settings", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ key: FORECAST_SETTING_KEY, value }) });
   123→      if (res.ok) {
   124→        toast.success(t("toast.saved"));
   125→        // Analytics (forecast) is React-Query cached — refetch it so the new
   126→        // threshold shows up the next time the Analytics view opens.
   127→        queryClient.invalidateQueries({ queryKey: ["analytics"] });
   128→        // Refetch settings so `savedThreshold` absorbs the new value, then
   129→        // drop the override (no flicker: the override holds until data lands).
   130→        await settings.refetch();
   131→        setThresholdOverride(null);
   132→      } else {
   133→        // save rejected — snap the slider back to the persisted value
   134→        setThresholdOverride(null);
   135→        toast.error(t("toast.save_failed"));
   136→      }
   137→    } catch {
   138→      setThresholdOverride(null);
   139→      toast.error(t("toast.save_failed"));
   140→    }
   141→  };
   142→  if (settings.isLoading || !org) return <Skeleton className="h-48 w-full" />;
   143→  return (
   144→    <div className="space-y-4 max-w-xl">
   145→      <Card><CardContent className="p-4 grid grid-cols-2 gap-3">
   146→        <div className="space-y-1 col-span-2"><Label className="text-xs">{t("settings.org.name")}</Label><Input value={name} onChange={(e) => setName(e.target.value)} /></div>
   147→        <div className="space-y-1"><Label className="text-xs">{t("settings.org.locale")}</Label>
   148→          <Select value={locale} onValueChange={setLocale}><SelectTrigger><SelectValue /></SelectTrigger><SelectContent>{["hy", "ru", "en"].map((l) => <SelectItem key={l} value={l}>{l === "hy" ? "Հայերեն" : l === "ru" ? "Русский" : "English"}</SelectItem>)}</SelectContent></Select>
   149→        </div>
   150→        <div className="space-y-1"><Label className="text-xs">{t("settings.org.timezone")}</Label><Input value={timezone} onChange={(e) => setTimezone(e.target.value)} /></div>
   151→        <div className="space-y-1"><Label className="text-xs">{t("settings.org.currency")}</Label><Input value={currency} onChange={(e) => setCurrency(e.target.value)} /></div>
   152→        <div className="col-span-2"><Button size="sm" onClick={save}><Save className="h-3.5 w-3.5 mr-1.5" />{t("common.save")}</Button></div>
   153→      </CardContent></Card>
   154→      {/* v0.23: forecast commit threshold (was hardcoded 60% in analytics-service) */}
   155→      <Card>
   156→        <CardContent className="p-4 space-y-3">
   157→          <div className="flex items-baseline justify-between gap-2">
   158→            <Label className="text-xs">{t("settings.org.forecast_commit")}</Label>
   159→            <span className="text-sm font-bold tabular-nums text-primary">{commitThreshold}%</span>
   160→          </div>
   161→          <Slider
   162→            value={[commitThreshold]}
   163→            min={10}
   164→            max={95}
   165→            step={5}
   166→            onValueChange={(v: number[]) => setThresholdOverride(v[0] ?? 60)}
   167→            onValueCommit={(v: number[]) => saveThreshold(v[0] ?? 60)}
   168→            aria-label={t("settings.org.forecast_commit")}
   169→          />
   170→          <p className="text-[11px] leading-relaxed text-muted-foreground">{t("settings.org.forecast_commit_hint")}</p>
   171→        </CardContent>
   172→      </Card>
   173→    </div>
   174→  );
   175→}

   177→function PipelineTab() {
   178→  const { t } = useLocale();
   179→  const settings = useSettings();
   180→  const createStage = useCreateStage();
   181→  const updateStage = useUpdateStage();
   182→  const delStage = useDeleteStage();
   183→  const [newStage, setNewStage] = useState<{ name: string; type: string; color: string }>({ name: "", type: "open", color: "#94a3b8" });
   184→
   185→  const sensors = useSensors(
   186→    useSensor(PointerSensor, { activationConstraint: { distance: 4 } }),
   187→    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
   188→  );
   189→
   190→  if (settings.isLoading) return <Skeleton className="h-48 w-full" />;
   191→  const pipelines = settings.data?.pipelines ?? [];
   192→
   193→  const addStage = async (pipelineId: string) => {
   194→    if (!newStage.name.trim()) return;
   195→    try {
   196→      await createStage.mutateAsync({ pipelineId, name: newStage.name, type: newStage.type, color: newStage.color });
   197→      toast.success("Stage added");
   198→      setNewStage({ name: "", type: "open", color: "#94a3b8" });
   199→    } catch (e) { toast.error((e as Error).message); }
   200→  };
   201→  const renameStage = async (id: string, name: string) => {
   202→    try { await updateStage.mutateAsync({ id, body: { name } }); } catch (e) { toast.error((e as Error).message); }
   203→  };
   204→  const recolorStage = async (id: string, color: string) => {
   205→    try { await updateStage.mutateAsync({ id, body: { color } }); } catch (e) { toast.error((e as Error).message); }
   206→  };
   207→  const removeStage = async (id: string) => {
   208→    try { await delStage.mutateAsync(id); toast.success("Stage deleted"); } catch (e) { toast.error((e as Error).message); }
   209→  };
   210→  const onDragEnd = async (pipelineId: string, event: DragEndEvent) => {
   211→    const { active, over } = event;
   212→    if (!over || active.id === over.id) return;
   213→    const p = pipelines.find((pp: any) => pp.id === pipelineId);
   214→    if (!p) return;
   215→    const oldIndex = p.stages.findIndex((s: any) => s.id === active.id);
   216→    const newIndex = p.stages.findIndex((s: any) => s.id === over.id);
   217→    if (oldIndex === -1 || newIndex === -1) return;
   218→    const reordered = arrayMove(p.stages as any[], oldIndex, newIndex) as any[];
   219→    // persist new positions
   220→    for (let i = 0; i < reordered.length; i++) {
   221→      if (reordered[i].position !== i) {
   222→        try { await updateStage.mutateAsync({ id: reordered[i].id, body: { position: i } }); } catch {}
   223→      }
   224→    }
   225→    toast.success("Stages reordered");
   226→  };
   227→
   228→  return (
   229→    <div className="space-y-3">
   230→      {pipelines.map((p: any) => (
   231→        <Card key={p.id}>
   232→          <CardHeader className="pb-2 flex flex-row items-center justify-between">
   233→            <CardTitle className="text-sm">{p.name}</CardTitle>
   234→            {p.isDefault && <Badge>Default</Badge>}
   235→          </CardHeader>
   236→          <CardContent className="pt-0 space-y-2">
   237→            <DndContext sensors={sensors} collisionDetection={closestCenter} onDragEnd={(e) => onDragEnd(p.id, e)}>
   238→              <SortableContext items={p.stages.map((s: any) => s.id)} strategy={verticalListSortingStrategy}>
   239→                <div className="space-y-1.5">
   240→                  {p.stages.map((s: any) => (
   241→                    <SortableStage
   242→                      key={s.id}
   243→                      stage={s}
   244→                      onRecolor={recolorStage}
   245→                      onRename={renameStage}
   246→                      onRemove={removeStage}
   247→                      updating={updateStage.isPending}
   248→                    />
   249→                  ))}
   250→                </div>
   251→              </SortableContext>
   252→            </DndContext>
   253→            {/* add new stage */}
   254→            <div className="flex items-center gap-2 pt-1 border-t">
   255→              <input
   256→                type="color"
   257→                value={newStage.color}
   258→                onChange={(e) => setNewStage((s) => ({ ...s, color: e.target.value }))}
   259→                className="h-6 w-6 rounded cursor-pointer border-0 bg-transparent p-0"
   260→              />
   261→              <Input
   262→                value={newStage.name}
   263→                onChange={(e) => setNewStage((s) => ({ ...s, name: e.target.value }))}
   264→                placeholder="New stage name…"
   265→                className="h-8 flex-1 text-sm"
   266→              />
   267→              <Select value={newStage.type} onValueChange={(v) => setNewStage((s) => ({ ...s, type: v }))}>
   268→                <SelectTrigger className="h-8 w-24 text-xs"><SelectValue /></SelectTrigger>
   269→                <SelectContent>
   270→                  <SelectItem value="open">open</SelectItem>
   271→                  <SelectItem value="won">won</SelectItem>
   272→                  <SelectItem value="lost">lost</SelectItem>
   273→                </SelectContent>
   274→              </Select>
   275→              <Button size="sm" variant="outline" onClick={() => addStage(p.id)} disabled={!newStage.name.trim() || createStage.isPending}>
   276→                <Plus className="h-3.5 w-3.5 mr-1" />{t("common.create")}
   277→              </Button>
   278→            </div>
   279→          </CardContent>
   280→        </Card>
   281→      ))}
   282→      <p className="text-xs text-muted-foreground px-1 flex items-center gap-1.5">
   283→        <GripVertical className="h-3 w-3" />
   284→        Drag the handle to reorder · rename inline · recolor via swatch · delete (blocked if leads are in the stage). Changes reflect immediately in the Kanban.
   285→      </p>
   286→    </div>
   287→  );
   288→}

   290→function SortableStage({ stage, onRecolor, onRename, onRemove, updating }: { stage: any; onRecolor: (id: string, c: string) => void; onRename: (id: string, n: string) => void; onRemove: (id: string) => void; updating: boolean }) {
   291→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id: stage.id });
   292→  const style = {
   293→    transform: CSS.Transform.toString(transform),
   294→    transition,
   295→    zIndex: isDragging ? 10 : "auto",
   296→  };
   297→  return (
   298→    <div ref={setNodeRef} style={style} className={cn("flex items-center gap-2 rounded-lg border px-2 py-1.5 bg-card", isDragging && "shadow-lg ring-2 ring-primary/30 opacity-90")}>
   299→      <button {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing text-muted-foreground hover:text-foreground touch-none" title="Drag to reorder">
   300→        <GripVertical className="h-4 w-4" />
   301→      </button>
   302→      <input
   303→        type="color"
   304→        value={stage.color ?? "#94a3b8"}
   305→        onChange={(e) => onRecolor(stage.id, e.target.value)}
   306→        className="h-6 w-6 rounded cursor-pointer border-0 bg-transparent p-0"
   307→        title="Stage color"
   308→      />
   309→      <input
   310→        defaultValue={stage.name}
   311→        onBlur={(e) => { if (e.target.value !== stage.name) onRename(stage.id, e.target.value); }}
   312→        className="flex-1 bg-transparent text-sm font-medium outline-none border-b border-transparent focus:border-primary"
   313→      />
   314→      <Badge variant="outline" className="text-[10px] px-1 py-0">{stage.type}</Badge>
   315→      {updating && <Loader2 className="h-3 w-3 animate-spin text-muted-foreground" />}
   316→      <button
   317→        onClick={() => onRemove(stage.id)}
   318→        className="text-muted-foreground hover:text-red-500 text-xs px-1"
   319→        title="Delete stage"
   320→      >✕</button>
   321→    </div>
   322→  );
   323→}

   355→function ScoringTab() {
   356→  const settings = useSettings();
   357→  const [rules, setRules] = useState<any[]>([]);
   358→  // local copy once loaded
   359→  useState(() => { if (settings.data?.scoring) setRules(settings.data.scoring); });

   361→  const list = rules.length ? rules : settings.data?.scoring ?? [];
   362→  const update = (idx: number, patch: Partial<any>) => setRules((cur) => cur.map((r, i) => i === idx ? { ...r, ...patch } : r));
   363→  const save = async () => {
   364→    try {
   365→      const res = await fetch("/api/v1/settings", { method: "PATCH", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ scoring: list.map((r: any) => ({ key: r.key, points: r.points, enabled: r.enabled })) }) });
   366→      if (res.ok) toast.success("Scoring saved"); else toast.error("Save failed");
   367→      settings.refetch();
   368→    } catch { toast.error("Save failed"); }
   369→  };
   370→  return (
   371→    <Card>
   372→      <CardContent className="p-4 space-y-2">
   373→        {list.map((r: any, i: number) => (
   374→          <div key={r.key} className="flex items-center gap-2">
   375→            <button onClick={() => update(i, { enabled: !r.enabled })} className={cn("h-5 w-5 rounded flex items-center justify-center border", r.enabled ? "bg-primary border-primary text-primary-foreground" : "bg-background")}><Check className="h-3 w-3" /></button>
   376→            <span className="flex-1 text-sm">{r.label}</span>
   377→            <Input type="number" value={r.points} onChange={(e) => update(i, { points: Number(e.target.value) })} className="w-20" />
   378→            <span className="text-xs text-muted-foreground">pts</span>
   379→          </div>
   380→        ))}
   381→        <Button size="sm" onClick={save}><Save className="h-3.5 w-3.5 mr-1.5" />Save scoring</Button>
   382→      </CardContent>
   383→    </Card>
   384→  );
   385→}
