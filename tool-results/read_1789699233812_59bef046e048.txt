   326→// ---------------------------------------------------------------------------
   327→// REVENUE FORECAST — stage-weighted pipeline (v0.18).
   328→// Empirical probabilities (from STAGE_CHANGE history) are shown as facts;
   329→// position-based fallbacks are visually muted and labeled as estimates.
   330→// ---------------------------------------------------------------------------
   331→
   332→type ForecastStage = {
   333→  stage: string;
   334→  color?: string | null;
   335→  count: number;
   336→  value: number;
   337→  probability: number;
   338→  empirical: boolean;
   339→  resolvedSamples: number;
   340→  weightedValue: number;
   341→};
   342→
   343→type Forecast = {
   344→  stages: ForecastStage[];
   345→  weightedTotal: number;
   346→  bestCase: number;
   347→  commit: number;
   348→  empiricalCoverage: number;
   349→};
   350→
   351→function ForecastCard({ forecast }: { forecast: Forecast }) {
   352→  const { t } = useLocale();
   353→  const { weightedTotal, bestCase, commit, stages, empiricalCoverage } = forecast;
   354→  // Position markers for the range bar (0..100% of bestCase).
   355→  const pct = (v: number) => (bestCase > 0 ? Math.min(100, Math.max(0, Math.round((v / bestCase) * 100))) : 0);
   356→  const hasData = stages.some((s) => s.count > 0);
   357→
   358→  return (
   359→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }}>
   360→      <Card className="overflow-hidden">
   361→        <CardHeader className="pb-3 bg-gradient-to-r from-primary/5 via-transparent to-transparent">
   362→          <CardTitle className="text-sm flex items-center gap-2">
   363→            <GaugeCircle className="h-4 w-4 text-primary" />
   364→            {t("analytics.forecast.title")}
   365→            <span className="ml-auto inline-flex items-center gap-1 rounded-full bg-muted px-2 py-0.5 text-[10px] font-medium text-muted-foreground normal-case">
   366→              <Info className="h-3 w-3" />
   367→              {t("analytics.forecast.coverage")}: {empiricalCoverage}%
   368→            </span>
   369→          </CardTitle>
   370→          <CardDescription className="text-xs">{t("analytics.forecast.subtitle")}</CardDescription>
   371→        </CardHeader>
   372→        <CardContent className="pt-3 space-y-4">
   373→          {!hasData ? (
   374→            <EmptyState icon={GaugeCircle} title={t("analytics.no_data")} hint={t("analytics.forecast.hint")} />
   375→          ) : (
   376→            <>
   377→              {/* headline: weighted forecast + range bar */}
   378→              <div className="flex flex-col sm:flex-row sm:items-end gap-4">
   379→                <div>
   380→                  <div className="text-[11px] uppercase tracking-wide text-muted-foreground">{t("analytics.forecast.weighted")}</div>
   381→                  <div className="text-3xl font-bold tabular-nums bg-gradient-to-r from-primary to-emerald-500 bg-clip-text text-transparent">
   382→                    {formatMoney(weightedTotal)}
   383→                  </div>
   384→                </div>
   385→                <div className="flex-1 min-w-[180px] sm:mb-2">
   386→                  {/* range bar: commit -> weighted -> best case */}
   387→                  <div className="relative h-3 rounded-full bg-muted overflow-visible">
   388→                    <div className="absolute inset-y-0 left-0 rounded-full bg-gradient-to-r from-emerald-500 via-primary to-violet-400" style={{ width: `${pct(weightedTotal)}%` }} />
   389→                    {[
   390→                      { v: commit, cls: "bg-emerald-500", lbl: t("analytics.forecast.commit") },
   391→                      { v: weightedTotal, cls: "bg-primary", lbl: t("analytics.forecast.weighted") },
   392→                      { v: bestCase, cls: "bg-violet-400", lbl: t("analytics.forecast.best") },
   393→                    ].map((m) => (
   394→                      <div key={m.lbl} className="absolute top-1/2 -translate-y-1/2 -translate-x-1/2 flex flex-col items-center" style={{ left: `${Math.max(2, Math.min(98, pct(m.v)))}%` }}>
   395→                        <span className={cn("h-4 w-1 rounded-full", m.cls)} />
   396→                        <span className="mt-1 text-[9px] text-muted-foreground whitespace-nowrap hidden sm:block">{m.lbl}</span>
   397→                      </div>
   398→                    ))}
   399→                  </div>
   400→                  <div className="flex justify-between mt-1 text-[10px] tabular-nums text-muted-foreground sm:mt-4">
   401→                    <span>{formatMoney(commit)}</span>
   402→                    <span>{formatMoney(bestCase)}</span>
   403→                  </div>
   404→                </div>
