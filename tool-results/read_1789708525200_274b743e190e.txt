    53→// ---------------------------------------------------------------------------
    54→// STAGE INACTIVITY configuration (per-stage thresholds keyed by STAGE ID)
    55→// ---------------------------------------------------------------------------
    56→
    57→function StageInactivityConfigCard() {
    58→  const { t } = useLocale();
    59→  const qc = useQueryClient();
    60→  const settings = useSettings();
    61→  const pipeline = usePipeline();
    62→  const [warning, setWarning] = useState<number>(DEFAULT_WARNING_BEFORE_HOURS);
    63→  const [thresholds, setThresholds] = useState<Record<string, number>>({});
    64→  const [saving, setSaving] = useState(false);
    65→  const [serverError, setServerError] = useState<string | null>(null);
    66→
    67→  // Real pipeline stages — Settings FOLLOWS the pipeline (Section 13): a new
    68→  // stage appears here automatically until it gets its own threshold.
    69→  const stages: { id: string; name: string; type: string }[] =
    70→    pipeline.data?.pipelines?.[0]?.stages ?? [];
    71→  const openStages = stages.filter((s) => s.type === "open");
    72→  const finalStages = stages.filter((s) => s.type !== "open");
    73→
    74→  // Load the stored config (keyed by stage ID — survives renames, Section 42).
    75→  useEffect(() => {
    76→    const rows = (settings.data?.settings as { key: string; value: unknown }[] | undefined) ?? [];
    77→    const row = rows.find((s) => s.key === STAGE_INACTIVITY_SETTING_KEY);
    78→    const parsed = row?.value != null ? parseStageInactivityConfig(row.value) : null;
    79→    if (parsed) {
    80→      setWarning(parsed.warningBeforeHours);
    81→      const map: Record<string, number> = {};
    82→      for (const [id, entry] of Object.entries(parsed.stages)) map[id] = entry.thresholdHours;
    83→      setThresholds(map);
    84→    }
    85→  }, [settings.data]);
    86→
    87→  // Value for a stage (input state), defaulting to the shared fallback.
    88→  const valueFor = (id: string) => thresholds[id] ?? DEFAULT_STAGE_INACTIVITY_HOURS;
    89→
    90→  // CLIENT-SIDE VALIDATION — same rules as the server (Section 15/16).
    91→  const validation = useMemo(
    92→    () =>
    93→      validateStageInactivityConfig({
    94→        warningBeforeHours: warning,
    95→        stages: Object.fromEntries(openStages.map((s) => [s.id, { thresholdHours: valueFor(s.id) }])),
    96→      }),
    97→    [warning, thresholds, openStages]
    98→  );
    99→
   100→  const save = async () => {
   101→    setSaving(true);
   102→    setServerError(null);
   103→    try {
   104→      const value = {
   105→        warningBeforeHours: warning,
   106→        // Only stages with an explicit value — untouched ones keep the fallback
   107→        // so a NEW stage still gets the default (Section 14).
   108→        stages: Object.fromEntries(
   109→          openStages
   110→            .filter((s) => thresholds[s.id] != null)
   111→            .map((s) => [s.id, { thresholdHours: thresholds[s.id] }])
   112→        ),
   113→      };
   114→      const res = await fetch("/api/v1/settings", {
   115→        method: "POST",
   116→        headers: { "Content-Type": "application/json" },
   117→        body: JSON.stringify({ key: STAGE_INACTIVITY_SETTING_KEY, value }),
   118→      });
   119→      if (res.ok) {
   120→        toast.success(t("stage.settings.title"));
   121→        // Thresholds changed → every stage health surface must recompute NOW.
   122→        await Promise.all([
   123→          qc.invalidateQueries({ queryKey: ["settings"] }),
   124→          qc.invalidateQueries({ queryKey: ["leads"] }),
   125→          qc.invalidateQueries({ queryKey: ["lead"] }),
   126→          qc.invalidateQueries({ queryKey: ["kanban"] }),
   127→          qc.invalidateQueries({ queryKey: ["dashboard"] }),
   128→        ]);
   129→      } else {
   130→        let msg = "Save failed";
   131→        try {
   132→          const body = await res.json();
   133→          msg = body?.error ?? msg;
   134→        } catch {}
   135→        setServerError(msg);
   136→        toast.error(msg);
   137→      }
   138→    } catch {
   139→      toast.error("Save failed");
   140→    } finally {
   141→      setSaving(false);
   142→    }
   143→  };
   144→
   145→  if (settings.isLoading || pipeline.isLoading) return <Skeleton className="h-64 w-full" />;
   146→
   147→  const now = Date.now();
   148→  const effectiveThresholds = Object.fromEntries(openStages.map((s) => [s.id, valueFor(s.id)]));
   149→  const previewConfig = { warningBeforeHours: warning, thresholds: effectiveThresholds };
   150→  const previewStage = openStages.find((s) => s.name === "Proposal")?.id ?? openStages[0]?.id ?? "";
   151→  const previewHours = effectiveThresholds[previewStage] ?? DEFAULT_STAGE_INACTIVITY_HOURS;
   152→
   153→  return (
   154→    <Card>
   155→      <CardHeader className="pb-2">
   156→        <CardTitle className="text-sm flex items-center gap-2"><Hourglass className="h-4 w-4" />{t("stage.settings.title")}</CardTitle>
   157→        <CardDescription className="text-xs">{t("stage.settings.desc")}</CardDescription>
   158→      </CardHeader>
   159→      <CardContent className="pt-0 space-y-3">
   160→        {/* global warning window */}
   161→        <div className="max-w-xs space-y-1">
   162→          <Label className="text-xs flex items-center gap-1.5" htmlFor="si-warning">
   163→            <span className="h-2 w-2 rounded-full bg-amber-500" />
   164→            {t("stage.settings.warning_before")}
   165→          </Label>
   166→          <Input
   167→            id="si-warning"
   168→            type="number"
   169→            min={1}
   170→            max={168}
   171→            step={1}
   172→            value={Number.isNaN(warning) ? "" : warning}
   173→            onChange={(e) => setWarning(e.target.value === "" ? NaN : Number(e.target.value))}
   174→            aria-invalid={!validation.ok}
   175→            className="h-8"
   176→          />
   177→          <p className="text-[10px] text-muted-foreground">{t("stage.aging")} · −{Number.isNaN(warning) ? "…" : humanizeDuration(warning * 60)}</p>
   178→        </div>
   179→
   180→        {/* per-stage thresholds — REAL pipeline stages, stacked rows (Section 97) */}
   181→        <div className="space-y-1.5">
   182→          {openStages.map((s) => (
   183→            <div key={s.id} className="flex items-center justify-between gap-3 rounded-lg border px-3 py-2">
   184→              <div className="min-w-0">
   185→                <p className="text-xs font-medium truncate">{s.name}</p>
   186→                <p className="text-[10px] text-muted-foreground">
   187→                  {thresholds[s.id] == null ? t("stage.settings.using_default", { hours: DEFAULT_STAGE_INACTIVITY_HOURS }) : `${t("stage.stale_after")} ${humanizeHours(thresholds[s.id])}`}
   188→                </p>
   189→              </div>
   190→              <div className="flex items-center gap-1.5 shrink-0">
   191→                <Input
   192→                  type="number"
   193→                  min={1}
   194→                  max={720}
   195→                  step={1}
   196→                  value={valueFor(s.id)}
   197→                  onChange={(e) => {
   198→                    const v = e.target.value === "" ? NaN : Number(e.target.value);
   199→                    setThresholds((cur) => ({ ...cur, [s.id]: v }));
   200→                  }}
   201→                  aria-label={`${s.name} — ${t("stage.settings.stale_after")}`}
   202→                  aria-invalid={!validation.ok}
   203→                  className="h-8 w-20 text-right"
   204→                />
   205→                <span className="text-[10px] text-muted-foreground w-10">{t("stage.settings.hours")}</span>
   206→              </div>
   207→            </div>
   208→          ))}
   209→          {finalStages.length > 0 && (
   210→            <div className="flex items-center gap-3 rounded-lg border border-dashed px-3 py-2">
   211→              <p className="text-xs text-muted-foreground">
   212→                {finalStages.map((s) => s.name).join(" · ")} — {t("stage.settings.not_monitored")}
   213→              </p>
   214→            </div>
   215→          )}
   216→        </div>
   217→
   218→        {/* inline validation errors (client) */}
   219→        {!validation.ok && (
   220→          <div role="alert" className="rounded-md border border-red-300 bg-red-50 dark:border-red-900 dark:bg-red-950/40 px-3 py-2">
   221→            <ul className="list-disc pl-4 text-xs text-red-700 dark:text-red-300 space-y-0.5">
   222→              {validation.errors.map((e) => <li key={e}>{e}</li>)}
   223→            </ul>
   224→          </div>
   225→        )}
   226→        {serverError && (
   227→          <div role="alert" className="rounded-md border border-red-300 bg-red-50 dark:border-red-900 dark:bg-red-950/40 px-3 py-2 text-xs text-red-700 dark:text-red-300">
   228→            {serverError}
   229→          </div>
   230→        )}
   231→
   232→        {/* Preview — the SAME StageHealthBadge used in the Lead List (Section 76). */}
   233→        <div className="rounded-lg border bg-muted/30 p-3">
   234→          <p className="text-xs text-muted-foreground mb-2">{t("stage.settings.preview")}</p>
   235→          <div className="flex items-center gap-2 flex-wrap">
   236→            <StageHealthBadge
   237→              leadStatus="OPEN"
   238→              stageId={previewStage}
   239→              stageName={openStages.find((s) => s.id === previewStage)?.name}
   240→              stageType="open"
   241→              stageEnteredAt={new Date(now - (previewHours - warning - 5) * 3_600_000)}
   242→              createdAt={new Date(now - (previewHours - warning - 5) * 3_600_000)}
   243→              config={previewConfig}
   244→            />
   245→            <StageHealthBadge
   246→              leadStatus="OPEN"
   247→              stageId={previewStage}
   248→              stageName={openStages.find((s) => s.id === previewStage)?.name}
   249→              stageType="open"
   250→              stageEnteredAt={new Date(now - (previewHours - warning + 1) * 3_600_000)}
   251→              createdAt={new Date(now - (previewHours - warning + 1) * 3_600_000)}
   252→              config={previewConfig}
   253→            />
   254→            <StageHealthBadge
   255→              leadStatus="OPEN"
   256→              stageId={previewStage}
   257→              stageName={openStages.find((s) => s.id === previewStage)?.name}
   258→              stageType="open"
   259→              stageEnteredAt={new Date(now - (previewHours + 2) * 3_600_000)}
   260→              createdAt={new Date(now - (previewHours + 2) * 3_600_000)}
   261→              config={previewConfig}
   262→            />
   263→          </div>
   264→        </div>
   265→
   266→        <Button size="sm" onClick={save} disabled={saving || !validation.ok}>
   267→          <Save className="h-3.5 w-3.5 mr-1.5" />
   268→          {saving ? "Saving…" : t("common.save")}
   269→        </Button>
   270→      </CardContent>
   271→    </Card>
   272→  );
   273→}

   285→  const [serverError, setServerError] = useState<string | null>(null);
   286→
   287→  useEffect(() => {
   288→    const rows = (settings.data?.settings as { key: string; value: unknown }[] | undefined) ?? [];
   289→    const row = rows.find((s) => s.key === FOLLOWUP_SETTING_KEY);
   290→    const parsed = row?.value != null ? parseFollowUpConfig(row.value) : null;
   291→    setConfig(parsed ?? DEFAULT_FOLLOWUP_SLA_CONFIG);
   292→  }, [settings.data]);
   293→
   294→  // CLIENT-SIDE VALIDATION — same rules as the server.
   295→  const validation = useMemo(() => validateFollowUpConfig(config), [config]);
   296→
   297→  const save = async () => {
   298→    setSaving(true);
   299→    setServerError(null);
   300→    try {
   301→      const res = await fetch("/api/v1/settings", {
   302→        method: "POST",
   303→        headers: { "Content-Type": "application/json" },
   304→        body: JSON.stringify({ key: FOLLOWUP_SETTING_KEY, value: validation.ok ? validation.config : config }),
   305→      });
   306→      if (res.ok) {
   307→        toast.success(t("followup.settings.title"));
   308→        await Promise.all([
   309→          qc.invalidateQueries({ queryKey: ["settings"] }),
   310→          qc.invalidateQueries({ queryKey: ["leads"] }),
   311→          qc.invalidateQueries({ queryKey: ["lead"] }),
   312→          qc.invalidateQueries({ queryKey: ["kanban"] }),

   316→        let msg = "Save failed";

   318→          const body = await res.json();

   320→        } catch {}
   321→        setServerError(msg);
   322→        toast.error(msg);
   323→      }

   325→      toast.error("Save failed");

   332→
   333→  const num = (v: string) => (v === "" ? NaN : Number(v));
   334→  const effective = validation.ok ? validation.config! : null;
   335→  const now = Date.now();
   336→
   337→  return (
   338→    <Card>
   339→      <CardHeader className="pb-2">
   340→        <CardTitle className="text-sm flex items-center gap-2"><Bell className="h-4 w-4" />{t("followup.settings.title")}</CardTitle>
   341→        <CardDescription className="text-xs">{t("followup.settings.desc")}</CardDescription>
   342→      </CardHeader>
   343→      <CardContent className="pt-0 space-y-3">
   344→        <div className="grid grid-cols-2 gap-3">
   345→          <div className="space-y-1">
   346→            <Label className="text-xs flex items-center gap-1.5" htmlFor="fu-warning">
   347→              <span className="h-2 w-2 rounded-full bg-amber-500" />
   348→              {t("followup.settings.warning_before")}
   349→            </Label>
   350→            <Input
   351→              id="fu-warning"
   352→              type="number"
   353→              min={0.5}
   354→              max={336}
   355→              step={0.5}
   356→              value={Number.isNaN(config.warningBeforeHours) ? "" : config.warningBeforeHours}
   357→              onChange={(e) => setConfig((c) => ({ ...c, warningBeforeHours: num(e.target.value) }))}
   358→              aria-invalid={!validation.ok}

   451→function FirstResponseCard() {

   455→  const [thresholds, setThresholds] = useState<SlaThresholds>(DEFAULT_SLA_THRESHOLDS);

   470→

   472→    setSaving(true);
   473→    setServerError(null);

   484→          qc.invalidateQueries({ queryKey: ["settings"] }),

   487→          qc.invalidateQueries({ queryKey: ["kanban"] }),
